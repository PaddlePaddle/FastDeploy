# SSH 连接并测试命令
# 在本地终端执行以下命令，或复制到服务器上执行

# ============================================
# 方法一：直接 SSH 执行（推荐）
# ============================================

ssh hj@10.26.43.21 << 'EOF'
# 设置工作目录
cd ~
PROJECT_DIR=~/FastDeploy
BRANCH=12
REPO_URL="https://github.com/kesmeey/FastDeploy.git"

# 克隆或更新代码
if [ -d "$PROJECT_DIR" ]; then
    echo "更新代码..."
    cd "$PROJECT_DIR"
    git fetch origin
    git checkout $BRANCH 2>/dev/null || git checkout -b $BRANCH origin/$BRANCH
    git pull origin $BRANCH
else
    echo "克隆代码..."
    git clone "$REPO_URL" "$PROJECT_DIR"
    cd "$PROJECT_DIR"
    git checkout $BRANCH
fi

# 显示当前提交
echo "当前提交:"
git log --oneline -1

# 设置环境变量
export PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION=python

# 检查并安装依赖
echo "检查依赖..."
python -c "import google.protobuf; print('protobuf:', google.protobuf.__version__)" 2>/dev/null || pip install "protobuf>=3.20.0,<4.0.0"
python -m pytest --version 2>/dev/null || pip install pytest pytest-asyncio

# 运行测试
echo "运行测试..."
cd "$PROJECT_DIR"
PROTOCOL_BUFFERS_PYTHON_IMPLEMENTATION=python python -m pytest tests/spec_decode/test_mtp_proposer.py -v
EOF
